Generate a personal key once, then add it to the WebhookExporter in your own project. This panel never uses sample measurements.



from ecotrace import EcoTrace

eco = EcoTrace(region_code="TR", carbon_limit=5.0)

@eco.track
def train_model(data):
    model = load_pretrained("bert-base")
    model.fit(data, epochs=10)
    return model

@eco.track
def preprocess(raw):
    return [tokenize(r) for r in raw]

@eco.track
def evaluate(model, test_data):
    return model.score(test_data)

# Pipeline
data = preprocess(load_data("dataset.csv"))
model = train_model(data)
print(f"Accuracy: {evaluate(model, data)}")
print(f"Carbon: {eco.total_carbon:.4f} gCO2")
eco.generate_pdf_report("audit.pdf")
